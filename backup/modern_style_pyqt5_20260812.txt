class ModernEffects:
    
    @staticmethod
    def add_shadow(widget, offset_x=0, offset_y=2, radius=8, color=None):
        shadow = QGraphicsDropShadowEffect()
        shadow.setOffset(offset_x, offset_y)
        shadow.setBlurRadius(radius)
        shadow.setColor(color if color else ModernColors.SHADOW)
        widget.setGraphicsEffect(shadow)
        return shadow
    
    @staticmethod
    def add_light_shadow(widget):
        return ModernEffects.add_shadow(
            widget,
            offset_x=ModernDimensions.SHADOW_OFFSET_X,
            offset_y=ModernDimensions.SHADOW_OFFSET_Y,
            radius=4,
            color=ModernColors.SHADOW_LIGHT
        )
    
    @staticmethod
    def add_card_shadow(widget):
        return ModernEffects.add_shadow(
            widget,
            offset_x=0,
            offset_y=4,
            radius=12,
            color=QColor(0, 0, 0, 80)
        )


class ModernAnimations:
    
    @staticmethod
    def create_hover_animation(widget, property_name='geometry', duration=150):
        animation = QPropertyAnimation(widget, property_name.encode())
        animation.setDuration(duration)
        animation.setEasingCurve(QEasingCurve.OutCubic)
        return animation
    
    @staticmethod
    def create_fade_in(widget, duration=200):
        animation = QPropertyAnimation(widget, b'windowOpacity')
        animation.setDuration(duration)
        animation.setStartValue(0.0)
        animation.setEndValue(1.0)
        animation.setEasingCurve(QEasingCurve.OutCubic)
        return animation


def apply_modern_style(widget):
    if isinstance(widget, QPushButton):
        widget.setStyleSheet(ModernStyle.get_button_style(primary=True))
        if not ModernStyle.is_macos():  # macOS 不需要額外陰影
            ModernEffects.add_light_shadow(widget)
    elif isinstance(widget, QLineEdit):
        widget.setStyleSheet(ModernStyle.get_input_style())
    elif isinstance(widget, QComboBox):
        widget.setStyleSheet(ModernStyle.get_combobox_style())
    elif isinstance(widget, QTextEdit):
        widget.setStyleSheet(ModernStyle.get_textedit_style())
    elif isinstance(widget, QGroupBox):
        widget.setStyleSheet(ModernStyle.get_groupbox_style())
        if not ModernStyle.is_macos():  # macOS 不需要額外陰影
            ModernEffects.add_card_shadow(widget)
    elif isinstance(widget, QTabWidget):
        widget.setStyleSheet(ModernStyle.get_tabwidget_style())
    elif isinstance(widget, QProgressBar):
        widget.setStyleSheet(ModernStyle.get_progressbar_style())
    elif isinstance(widget, QScrollArea):
        widget.setStyleSheet(ModernStyle.get_scrollarea_style())
    elif isinstance(widget, QMenu):
        widget.setStyleSheet(ModernStyle.get_menu_style())
    elif isinstance(widget, QToolBar):
        widget.setStyleSheet(ModernStyle.get_toolbar_style())
    elif isinstance(widget, QToolButton):
        widget.setStyleSheet(ModernStyle.get_toolbar_style())
    elif isinstance(widget, QDockWidget):
        widget.setStyleSheet(ModernStyle.get_dockwidget_style())
    elif isinstance(widget, QTableWidget):
        widget.setStyleSheet(ModernStyle.get_table_style())
    elif isinstance(widget, QCheckBox):
        widget.setStyleSheet(ModernStyle.get_checkbox_style())
    elif isinstance(widget, QRadioButton):
        widget.setStyleSheet(ModernStyle.get_radiobutton_style())
    elif isinstance(widget, (QSpinBox, QDoubleSpinBox)):
        widget.setStyleSheet(ModernStyle.get_spinbox_style())
    elif isinstance(widget, QDialog):
        widget.setStyleSheet(ModernStyle.get_dialog_style())
    elif isinstance(widget, QLabel):
        widget.setStyleSheet(ModernStyle.get_label_style())

    palette.setColor(QPalette.BrightText, QColor(ModernColors.ERROR))
    palette.setColor(QPalette.Link, QColor(ModernColors.PRIMARY))
    palette.setColor(QPalette.Highlight, QColor(ModernColors.PRIMARY))
    palette.setColor(QPalette.HighlightedText, QColor(ModernColors.TEXT_ON_PRIMARY))
    return palette


def apply_global_modern_style(app):
    app.setStyle('Fusion')
    app.setPalette(get_modern_palette())
    app.setStyleSheet(ModernStyle.get_global_style())
    
    # 如果是 macOS，則進一步優化樣式
    if ModernStyle.is_macos():
        # macOS 特定的全局樣式調整
        macos_styles = """
        QMenuBar {
            background-color: transparent;
            spacing: 3px;
        }
        QMenuBar::item {
            background: transparent;
            padding: 4px 8px;
            border-radius: 4px;
        }
        QMenuBar::item:selected {
            background: #404040;
        }
        QMenuBar::item:pressed {
            background: #7B9E87;
        }
        QStatusBar {
            background-color: #2D2D2D;
            border-top: 1px solid #4A4A4A;
        }
        """
        app.setStyleSheet(app.styleSheet() + macos_styles)
